alter table iwasp_messages enable row level security; create table if not exists monthly_bonus_log ( referrer text not null, referred text not null, month text not null, created_at timestamptz default now(), primary key (referrer, referred, month) ); alter table monthly_bonus_log enable row level security; create or replace function iwasp_get_user(p_tg text, p_name text, p_ref text default null) returns numeric language plpgsql security definer as $$ declare b numeric; bonus numeric := 200; tier text; begin if p_ref is not null and p_ref <> p_tg then select referral_tier into tier from users where telegram_id = p_ref; if tier = 'SUPER' then bonus := 300; end if; end if; insert into users (telegram_id, username, referred_by, referral_bonus) values (p_tg, p_name, case when p_ref is not null and p_ref <> p_tg then p_ref end, bonus) on conflict (telegram_id) do nothing; select balance into b from users where telegram_id = p_tg; return coalesce(b, 0); end; $$; create or replace function iwasp_referrals(p_tg text) returns json language sql security definer as $$ select json_build_object( 'invited', count(*), 'approved', count(*) filter (where coalesce(referral_paid, false)), 'pending', count(*) filter (where not coalesce(referral_paid, false)), 'approved_coins', coalesce(sum(coalesce(referral_bonus, 200)) filter (where coalesce(referral_paid, false)), 0), 'pending_coins', coalesce(sum(coalesce(referral_bonus, 200)) filter (where not coalesce(referral_paid, false)), 0), 'tier', coalesce((select referral_tier from users where telegram_id = p_tg), 'STANDARD') ) from users where referred_by = p_tg; $$; create or replace function iwasp_referral_list(p_tg text) returns json language sql security definer as $$ select coalesce( json_agg(json_build_object( 'name', coalesce(username, 'User'), 'id', right(telegram_id, 4), 'approved', coalesce(referral_paid, false), 'bonus', coalesce(referral_bonus, 200) )), '[]'::json) from users where referred_by = p_tg; $$; create or replace function credit_offer(p_user text, p_offer text, p_payout numeric) returns void language plpgsql security definer as $$ declare total numeric; ref text; bonus numeric; begin insert into conversions (user_id, offer_id, payout) values (p_user, p_offer, p_payout); update users set balance = coalesce(balance, 0) + p_payout where telegram_id = p_user; select sum(payout) into total from conversions where user_id = p_user; select referred_by, coalesce(referral_bonus, 200) into ref, bonus from users where telegram_id = p_user and coalesce(referral_paid, false) = false; if ref is not null and total >= 700 then update users set balance = coalesce(balance, 0) + bonus where telegram_id = ref; update users set referral_paid = true where telegram_id = p_user; end if; end; $$; create or replace function iwasp_get_messages() returns json language sql security definer as $$ select coalesce(json_agg(m order by m.id desc), '[]'::json) from (select id, title, body, created_at from iwasp_messages order by id desc limit 20) m; $$; create or replace function run_monthly_bonus(p_month text) returns integer language plpgsql security definer as $$ declare r record; n integer := 0; start_ts timestamptz; end_ts timestamptz; begin start_ts := (p_month || '-01')::timestamptz; end_ts := start_ts + interval '1 month'; for r in select u.referred_by as referrer, u.telegram_id as referred from users u join users s on s.telegram_id = u.referred_by where s.referral_tier = 'SUPER' and (select coalesce(sum(payout), 0) from conversions c where c.user_id = u.telegram_id and c.created_at >= start_ts and c.created_at < end_ts) >= 3000 loop begin insert into monthly_bonus_log (referrer, referred, month) values (r.referrer, r.referred, p_month); update users set balance = coalesce(balance, 0) + 100 where telegram_id = r.referrer; n := n + 1; exception when unique_violation then null; end; end loop; return n; end; $$;
